function cadastroSemErros(formId) { return !Array.from(document.querySelectorAll("#" + formId + " .text-danger")) .some(function (el) { return el.textContent.trim() !== ""; }); } function fecharModalCadastroSeValido(data, modalId, formId) { if (data.status !== "success" || !cadastroSemErros(formId)) return; bootstrap.Modal.getOrCreateInstance(document.getElementById(modalId)).hide(); } function abrirModalEditarCadastro(data) { if (data.status === "success") bootstrap.Modal.getOrCreateInstance(document.getElementById("modalEditarCadastro")).show(); } function fecharModalIncluirCadastro(data) { fecharModalCadastroSeValido(data, "modalIncluirCadastro", "formIncluirCadastro"); } function fecharModalEditarCadastro(data) { fecharModalCadastroSeValido(data, "modalEditarCadastro", "formEditarCadastro"); }
